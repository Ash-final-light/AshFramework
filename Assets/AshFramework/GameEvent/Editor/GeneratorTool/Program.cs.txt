using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.CSharp.Syntax;

// 本文件是独立 .NET 工具源码，不参与 Unity 编译。
internal static class Program
{
    private const string Header = "// <auto-generated by AshFramework Event Bridge>\n// 请修改事件接口，然后执行 Tools/Generate Event Bridge。\n";
    private const string ManifestName = ".eventbridge-manifest.json";
    private const string Api = "global::AshFramework.Events.";
    private static readonly JsonSerializerOptions JsonOptions = new JsonSerializerOptions { WriteIndented = true };

    private sealed record Contract(string Name, string Namespace, string Assembly, string Output, string Source,
        InterfaceDeclarationSyntax Syntax, CompilationUnitSyntax Unit);
    private sealed record Change(string Path, string Operation, string Content);
    private sealed record Manifest(int Version, string[] Files);

    public static int Main(string[] args)
    {
        try
        {
            if (args.Contains("--self-test")) { SelfTest(); return 0; }
            var project = Path.GetFullPath(Option(args, "--project") ?? throw new ArgumentException("缺少 --project"));
            if (!Directory.Exists(Path.Combine(project, "Assets"))) throw new ArgumentException("目标不是 Unity 项目");
            var defines = (Option(args, "--defines") ?? "UNITY_EDITOR;UNITY_INCLUDE_TESTS").Split(';', StringSplitOptions.RemoveEmptyEntries);
            var changes = Plan(project, defines);
            var report = new { project, changes = changes.Select(c => new { path = c.Path, operation = c.Operation }).ToArray() };
            if (args.Contains("--apply"))
            {
                if (changes.Any(c => c.Operation == "delete") && !args.Contains("--allow-delete"))
                    throw new InvalidOperationException("存在过期生成文件；检查预览后显式传入 --allow-delete。");
                foreach (var change in changes)
                {
                    if (change.Operation == "delete") File.Delete(change.Path);
                    else
                    {
                        Directory.CreateDirectory(Path.GetDirectoryName(change.Path));
                        File.WriteAllText(change.Path, change.Content, new UTF8Encoding(false));
                    }
                }
            }
            Console.WriteLine(JsonSerializer.Serialize(report, JsonOptions));
            return args.Contains("--check") && changes.Count > 0 ? 2 : 0;
        }
        catch (Exception error) { Console.Error.WriteLine(error.Message); return 1; }
    }

    private static string Option(string[] args, string key)
    {
        var index = Array.IndexOf(args, key);
        return index >= 0 && index + 1 < args.Length ? args[index + 1] : null;
    }

    private static List<Change> Plan(string project, string[] defines)
    {
        var assets = Path.Combine(project, "Assets");
        var contracts = new List<Contract>();
        foreach (var file in Directory.EnumerateFiles(assets, "*.cs", SearchOption.AllDirectories).OrderBy(f => f, StringComparer.Ordinal))
        {
            var relative = Path.GetRelativePath(assets, file).Replace('\\', '/');
            if (relative.Split('/').Any(p => p == "Editor" || p == "Generated" || p.StartsWith('.') || p.EndsWith('~'))) continue;
            var tree = CSharpSyntaxTree.ParseText(File.ReadAllText(file),
                new CSharpParseOptions(LanguageVersion.CSharp9, preprocessorSymbols: defines), file);
            var unit = tree.GetCompilationUnitRoot();
            foreach (var syntax in unit.DescendantNodes().OfType<InterfaceDeclarationSyntax>().Where(IsEvent))
            {
                var errors = tree.GetDiagnostics().Where(d => d.Severity == DiagnosticSeverity.Error).ToArray();
                if (errors.Length > 0) throw new InvalidOperationException(string.Join("\n", errors.Select(e => e.ToString())));
                Validate(syntax, file);
                var ns = string.Join(".", syntax.Ancestors().OfType<BaseNamespaceDeclarationSyntax>().Reverse().Select(n => n.Name.ToString()));
                var root = FindAssemblyRoot(project, file, out var assembly);
                contracts.Add(new Contract(syntax.Identifier.ValueText, ns, assembly,
                    Path.Combine(root, "Generated", "EventBridge"), file, syntax, unit));
            }
        }
        var fullNames = new HashSet<string>(StringComparer.Ordinal);
        foreach (var contract in contracts)
            if (!fullNames.Add(FullName(contract))) throw new InvalidOperationException("重复事件接口：" + FullName(contract));
        foreach (var group in contracts.GroupBy(c => c.Namespace))
            if (group.Select(c => c.Assembly).Distinct().Count() > 1)
                throw new InvalidOperationException("同一事件命名空间跨多个程序集会导致 GameEventHelper 冲突，请使用不同命名空间：" + group.Key);

        var expected = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
        var roots = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        foreach (var contract in contracts.OrderBy(FullName, StringComparer.Ordinal))
        {
            roots.Add(contract.Output);
            expected.Add(Path.Combine(contract.Output, contract.Name + "." + Hash(FullName(contract)) + ".g.cs"), Generate(contract));
        }
        foreach (var group in contracts.GroupBy(c => (c.Output, c.Namespace)))
        {
            expected.Add(Path.Combine(group.Key.Output, "EventBridge.Init." + Hash(group.Key.Namespace) + ".g.cs"),
                GenerateInit(group.Key.Namespace, group.OrderBy(FullName, StringComparer.Ordinal)));
        }
        foreach (var path in Directory.EnumerateFiles(assets, ManifestName, SearchOption.AllDirectories))
            roots.Add(Path.GetDirectoryName(path));

        var changes = new List<Change>();
        foreach (var root in roots.OrderBy(r => r, StringComparer.Ordinal))
        {
            var manifestPath = Path.Combine(root, ManifestName);
            var previous = File.Exists(manifestPath) ? JsonSerializer.Deserialize<Manifest>(File.ReadAllText(manifestPath)) : null;
            if (previous != null && (previous.Version != 1 || previous.Files == null))
                throw new InvalidOperationException("无效生成清单：" + manifestPath);
            var outputs = expected.Where(e => Path.GetDirectoryName(e.Key) == root).OrderBy(e => e.Key, StringComparer.Ordinal).ToArray();
            var ownedNames = outputs.Select(e => Path.GetFileName(e.Key)).ToArray();
            foreach (var name in previous?.Files ?? Array.Empty<string>())
            {
                if (name != Path.GetFileName(name) || !name.EndsWith(".g.cs", StringComparison.Ordinal))
                    throw new InvalidOperationException("清单包含非法路径：" + manifestPath);
                var path = Path.Combine(root, name);
                if (!expected.ContainsKey(path) && File.Exists(path))
                {
                    EnsureOwned(path);
                    changes.Add(new Change(path, "delete", null));
                }
            }
            foreach (var output in outputs)
            {
                if (File.Exists(output.Key)) EnsureOwned(output.Key);
                AddIfChanged(changes, output.Key, output.Value);
            }
            if (outputs.Length > 0 || previous != null)
                AddIfChanged(changes, manifestPath, JsonSerializer.Serialize(new Manifest(1, ownedNames), JsonOptions) + "\n");
        }
        return changes;
    }

    private static void AddIfChanged(List<Change> changes, string path, string text)
    {
        if (!File.Exists(path)) changes.Add(new Change(path, "create", text));
        else if (File.ReadAllText(path) != text) changes.Add(new Change(path, "update", text));
    }

    private static void EnsureOwned(string path)
    {
        if (!File.ReadAllText(path).StartsWith(Header, StringComparison.Ordinal))
            throw new InvalidOperationException("拒绝覆盖或删除非本工具生成文件：" + path);
    }

    private static string FindAssemblyRoot(string project, string file, out string assembly)
    {
        var assets = Path.Combine(project, "Assets");
        for (var directory = Path.GetDirectoryName(file); directory != null; directory = Path.GetDirectoryName(directory))
        {
            if (Directory.EnumerateFiles(directory, "*.asmref").Any())
                throw new InvalidOperationException("暂不支持 asmref，请将事件接口放在明确的 asmdef 目录：" + file);
            var definitions = Directory.EnumerateFiles(directory, "*.asmdef").ToArray();
            if (definitions.Length > 1) throw new InvalidOperationException("目录包含多个 asmdef：" + directory);
            if (definitions.Length == 1)
            {
                using var json = JsonDocument.Parse(File.ReadAllText(definitions[0]));
                assembly = json.RootElement.GetProperty("name").GetString();
                return directory;
            }
            if (directory.Equals(assets, StringComparison.OrdinalIgnoreCase)) break;
        }
        assembly = "Assembly-CSharp";
        return Path.Combine(assets, "AshFramework", "GameEvent");
    }

    private static bool IsEvent(InterfaceDeclarationSyntax syntax) => syntax.AttributeLists.SelectMany(l => l.Attributes)
        .Any(a => a.Name.ToString().Split('.').Last() is "EventInterface" or "EventInterfaceAttribute");

    private static void Validate(InterfaceDeclarationSyntax syntax, string file)
    {
        void Fail(string text) => throw new InvalidOperationException(file + ": " + syntax.Identifier.ValueText + "：" + text);
        if (!syntax.Modifiers.Any(SyntaxKind.PublicKeyword)) Fail("事件接口必须 public");
        if (syntax.Ancestors().Any(n => n is TypeDeclarationSyntax)) Fail("不支持嵌套事件接口");
        if (syntax.TypeParameterList != null || syntax.BaseList != null || syntax.Modifiers.Any(SyntaxKind.PartialKeyword))
            Fail("不支持泛型、继承或 partial 事件接口");
        if (syntax.Members.Any(m => m is not MethodDeclarationSyntax)) Fail("事件接口仅允许普通方法");
        var names = new HashSet<string>(StringComparer.Ordinal);
        foreach (var method in syntax.Members.OfType<MethodDeclarationSyntax>())
        {
            if (method.ReturnType.ToString() != "void" || method.TypeParameterList != null || method.Body != null || method.ExpressionBody != null || method.Modifiers.Count != 0)
                Fail("方法必须是无实现的非泛型 void 通知：" + method.Identifier.ValueText);
            if (!names.Add(method.Identifier.ValueText)) Fail("不支持事件方法重载：" + method.Identifier.ValueText);
            if (method.ParameterList.Parameters.Count > 6) Fail("最多支持六个参数");
            if (method.ParameterList.Parameters.Any(p => p.Modifiers.Count > 0 || p.Type is PointerTypeSyntax || p.Type?.ToString().Contains('*') == true))
                Fail("不支持 ref/out/in/params 或指针参数：" + method.Identifier.ValueText);
        }
    }

    private static string FullName(Contract contract) => string.IsNullOrEmpty(contract.Namespace) ? contract.Name : contract.Namespace + "." + contract.Name;
    private static string Hash(string value) => Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(value))).Substring(0, 12).ToLowerInvariant();
    private static string Literal(string value) => SymbolDisplay.FormatLiteral(value, true);

    private static string Generate(Contract contract)
    {
        var name = contract.Syntax.Identifier.Text;
        var methods = contract.Syntax.Members.OfType<MethodDeclarationSyntax>().OrderBy(m => m.Identifier.ValueText, StringComparer.Ordinal).ToArray();
        var code = new StringBuilder();
        code.AppendLine("public static class " + name + "_Event\n{");
        foreach (var method in methods)
            code.AppendLine("    public static readonly int " + method.Identifier.Text + " = " + Api + "RuntimeId.ToRuntimeId(" + Literal(FullName(contract) + "." + method.Identifier.ValueText) + ");");
        code.AppendLine("}\n");
        code.AppendLine("public sealed class " + name + "_Gen : global::" + FullName(contract) + "\n{");
        code.AppendLine("    private readonly " + Api + "EventDispatcher _dispatcher;");
        code.AppendLine("    public " + name + "_Gen(" + Api + "EventDispatcher dispatcher)\n    {");
        code.AppendLine("        _dispatcher = dispatcher ?? throw new global::System.ArgumentNullException(nameof(dispatcher));");
        code.AppendLine("        " + Api + "GameEvent.EventMgr.RegWrapInterface<global::" + FullName(contract) + ">(this);\n    }");
        foreach (var method in methods)
        {
            var parameters = string.Join(", ", method.ParameterList.Parameters.Select(p => p.Type + " " + p.Identifier.Text));
            var types = method.ParameterList.Parameters.Select(p => p.Type.ToString()).ToArray();
            var generic = types.Length == 0 ? "" : "<" + string.Join(", ", types) + ">";
            var arguments = string.Join(", ", method.ParameterList.Parameters.Select(p => p.Identifier.Text));
            code.AppendLine("    public void " + method.Identifier.Text + "(" + parameters + ")\n    {");
            code.AppendLine("        _dispatcher.Send" + generic + "(" + name + "_Event." + method.Identifier.Text + (arguments.Length == 0 ? "" : ", " + arguments) + ");\n    }");
        }
        code.AppendLine("}\n");
        var groupName = name.Length > 1 && name[0] == 'I' && char.IsUpper(name[1]) ? name.Substring(1) : name;
        code.AppendLine("public static partial class GameEventHelper\n{");
        foreach (var operation in new[] { "Add", "Remove" })
        {
            code.AppendLine("    public static partial class " + operation + "Event\n    {\n        public static class " + groupName + "\n        {");
            foreach (var method in methods)
            {
                var types = method.ParameterList.Parameters.Select(p => p.Type.ToString()).ToArray();
                var generic = types.Length == 0 ? "" : "<" + string.Join(", ", types) + ">";
                code.AppendLine("            public static void " + method.Identifier.Text + "(global::System.Action" + generic + " action)\n            {");
                code.AppendLine("                " + Api + "GameEvent." + operation + "EventListener(" + name + "_Event." + method.Identifier.Text + ", action);\n            }");
            }
            code.AppendLine("        }\n    }");
        }
        code.AppendLine("}");
        var result = new StringBuilder(Header);
        foreach (var usingDirective in contract.Unit.Usings) result.AppendLine(usingDirective.ToString());
        var namespaces = contract.Syntax.Ancestors().OfType<BaseNamespaceDeclarationSyntax>().Reverse().ToArray();
        foreach (var ns in namespaces)
        {
            result.AppendLine("namespace " + ns.Name + "\n{");
            foreach (var usingDirective in ns.Usings) result.AppendLine(usingDirective.ToString());
        }
        result.Append(code);
        foreach (var ns in namespaces) result.AppendLine("}");
        return result.ToString().Replace("\r\n", "\n");
    }

    private static string GenerateInit(string ns, IEnumerable<Contract> contracts)
    {
        var result = new StringBuilder(Header);
        if (ns.Length > 0) result.AppendLine("namespace " + ns + "\n{");
        result.AppendLine("public static partial class GameEventHelper\n{\n    public static void Init()\n    {");
        foreach (var contract in contracts)
        {
            result.AppendLine("        if (" + Api + "GameEvent.EventMgr.GetInterface<global::" + FullName(contract) + ">() == null)");
            result.AppendLine("            new " + contract.Name + "_Gen(" + Api + "GameEvent.EventMgr.GetDispatcher());");
        }
        result.AppendLine("    }\n}");
        if (ns.Length > 0) result.AppendLine("}");
        return result.ToString();
    }

    private static void SelfTest()
    {
        var root = Path.Combine(Path.GetTempPath(), "AshEventBridge-" + Guid.NewGuid().ToString("N"));
        var sourceDirectory = Path.Combine(root, "Assets", "Contracts");
        Directory.CreateDirectory(sourceDirectory);
        var source = Path.Combine(sourceDirectory, "IExample.cs");
        var count = 0;
        void Assert(bool condition, string description)
        {
            if (!condition) throw new Exception("FAIL: " + description);
            Console.WriteLine("PASS: " + description); count++;
        }
        void Apply(List<Change> changes)
        {
            foreach (var change in changes)
            {
                if (change.Operation == "delete") File.Delete(change.Path);
                else { Directory.CreateDirectory(Path.GetDirectoryName(change.Path)); File.WriteAllText(change.Path, change.Content); }
            }
        }
        var defines = new[] { "UNITY_INCLUDE_TESTS" };
        File.WriteAllText(source, "using AshFramework.Events; using Value = System.Int32; namespace Feature { [EventInterface(EEventGroup.GroupLogic)] public interface IExample { void Update(Value value); } }");
        var initial = Plan(root, defines);
        Assert(initial.Count == 3, "首次生成代理、初始化与清单");
        var generated = initial.Where(c => c.Path.EndsWith(".cs")).Select(c => CSharpSyntaxTree.ParseText(c.Content)).ToArray();
        var references = ((string)AppContext.GetData("TRUSTED_PLATFORM_ASSEMBLIES")).Split(Path.PathSeparator)
            .Select(p => MetadataReference.CreateFromFile(p)).Append(MetadataReference.CreateFromFile(typeof(AshFramework.Events.GameEvent).Assembly.Location));
        var compilation = CSharpCompilation.Create("BridgeFixture", generated.Append(CSharpSyntaxTree.ParseText(File.ReadAllText(source))), references,
            new CSharpCompilationOptions(OutputKind.DynamicallyLinkedLibrary));
        using (var output = new MemoryStream())
        {
            var result = compilation.Emit(output);
            Assert(result.Success, "生成源码可编译，保留 using 别名：" + string.Join(";", result.Diagnostics.Where(d => d.Severity == DiagnosticSeverity.Error)));
        }
        Apply(initial);
        Assert(Plan(root, defines).Count == 0, "重复生成零变化");
        File.WriteAllText(source, "using AshFramework.Events; namespace Feature { [EventInterface(EEventGroup.GroupLogic)] public interface IExample { void Update(string value); void Added(); } }");
        Assert(Plan(root, defines).Count(c => c.Operation == "update") == 1, "接口改参数及新增方法不依赖旧程序集");
        Apply(Plan(root, defines));
        File.WriteAllText(source, "using AshFramework.Events; namespace Feature { [EventInterface(EEventGroup.GroupLogic)] public interface IExample { void Update(int value); void Update(string value); } }");
        var rejected = false;
        try { Plan(root, defines); } catch (InvalidOperationException) { rejected = true; }
        Assert(rejected, "不支持的重载明确拒绝且不写文件");
        File.Delete(source);
        var removed = Plan(root, defines);
        Assert(removed.Count(c => c.Operation == "delete") == 2, "删除接口只清理清单拥有的生成源码");
        Apply(removed);
        Assert(Plan(root, defines).Count == 0, "清理后重复生成零变化");
        File.WriteAllText(source, "using AshFramework.Events; namespace Other { [EventInterface(EEventGroup.GroupLogic)] public interface IExample { void Ping(); } }");
        var recreate = Plan(root, defines);
        var target = recreate.First(c => c.Path.EndsWith(".cs"));
        Directory.CreateDirectory(Path.GetDirectoryName(target.Path));
        File.WriteAllText(target.Path, "// 用户文件");
        rejected = false;
        try { Plan(root, defines); } catch (InvalidOperationException) { rejected = true; }
        Assert(rejected, "拒绝覆盖同路径的用户文件");
        Console.WriteLine("SELF_TEST_PASSED=" + count);
        // 临时验证目录保留便于追溯，不清理任何用户目录。
    }
}
